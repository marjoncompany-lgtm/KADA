Use GitHub's free cloud runners to build an independent, raw .apk binary for free without any local Android SDK.

Workflow: .github/workflows/build-apk.yml 
name: Build KADA 5.0mins APK
on: [push, workflow_dispatch]
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-java@v4
        with: { java-version: '17', distribution: 'temurin' }
      - uses: actions/setup-node@v4
        with: { node-version: '20' }
      - run: npm install -g @bubblewrap/cli
      - run: |
          mkdir -p android-build
          npx @bubblewrap/cli init --manifest=https://${{ github.event.repository.name }}.pages.dev/manifest.json --directory=android-build
          cd android-build && npx @bubblewrap/cli build
      - uses: actions/upload-artifact@v4
        with: { name: KADA-5.0mins-Release-APK, path: android-build/*.apk }
